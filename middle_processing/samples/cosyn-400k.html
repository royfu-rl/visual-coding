import numpy as np
import matplotlib.pyplot as plt
import io
from PIL import Image

def generate_plot(data):
    plt.style.use('seaborn-v0_8-whitegrid')
    
    # Prepare data
    x = data['Wait_Time_Minutes']
    y = data['Frequency']
    
    # Create figure and axis
    fig, ax = plt.subplots(figsize=(12, 8))
    
    # Create hexbin plot
    hb = ax.hexbin(x, y, gridsize=15, cmap='viridis', edgecolors='white')
    
    # Customize plot
    ax.set_title('Patient Wait Times vs Frequency', fontsize=20, fontweight='bold', pad=20)
    ax.set_xlabel('Wait Time (Minutes)', fontsize=14, labelpad=15)
    ax.set_ylabel('Frequency', fontsize=14, labelpad=15)
    
    # Add colorbar
    cbar = plt.colorbar(hb, ax=ax)
    cbar.set_label('Count', rotation=270, labelpad=20, fontsize=12)
    
    # Customize ticks
    ax.tick_params(axis='both', which='major', labelsize=12)
    
    # Add text annotations for departments
    for _, row in data.iterrows():
        ax.annotate(row['Department'], (row['Wait_Time_Minutes'], row['Frequency']),
                    xytext=(5, 5), textcoords='offset points', fontsize=8,
                    bbox=dict(boxstyle="round,pad=0.3", fc="white", ec="gray", alpha=0.8))
    
    # Adjust layout
    plt.tight_layout()
    
    # Save plot to BytesIO object
    buf = io.BytesIO()
    plt.savefig(buf, format='png', dpi=300, bbox_inches='tight')
    buf.seek(0)
    
    # Convert to PIL Image
    img = Image.open(buf)
    
    return img
