      // section (#anchor), reduced motion, for crawlers or audit tools (Lighthouse, which
      // sees the page as crawlers do). Without JS it's all just there.
      (() => {
        const root = document.documentElement;
        root.classList.add('js');
        // Trying out the letter without its reading focus: felixhaas.io/?focus=off (global.css).
        if (new URLSearchParams(location.search).get('focus') === 'off') root.classList.add('focus-off');
        const nav = performance.getEntriesByType('navigation')[0];
        const quiet =
          matchMedia('(prefers-reduced-motion: reduce)').matches || /bot|crawl|spider|slurp|lighthouse/i.test(navigator.userAgent);
        if (nav?.type === 'back_forward' || location.hash || quiet) return;

        history.scrollRestoration = 'manual';
        root.classList.add('splash');
        // Left alone, it plays out (global.css): the opening sentence appears word by word,
        // then at 2.9s the letter comes in, the dot with it, and once it has arrived (2.6s
        // later) colors go back to their quick reading transitions.
        const events = ['wheel', 'touchstart', 'keydown', 'pointerdown', 'scroll'];
        let settling = 0;
        const settle = () => {
          events.forEach((type) => removeEventListener(type, rush));
          root.classList.add('settled');
        };
        const reveal = () => {
          root.classList.add('revealed');
          settling = setTimeout(settle, 2600);
        };
        // But it never keeps you waiting: the first scroll, touch or key press, at any point,
        // brings what's still to come at once: whatever waits skips its wait, and whatever
        // is under way finishes within 0.45s (the letter's way up, if it hasn't set off,
        // within 0.5s on a quick ease-out).
        const rush = () => {
          events.forEach((type) => removeEventListener(type, rush));
          clearTimeout(revealing);
          clearTimeout(settling);
          opened = true;
          root.classList.add('opening', 'revealed', 'rushed');

          const opening = document

            );

            const { delay, activeDuration, localTime } = animation.effect.getComputedTiming();

          });
